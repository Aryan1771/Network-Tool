How to Execute NS-2 on Ubuntu

1. Install NS-2 and Network Animator (NAM): 
sudo apt update
sudo apt install ns2 nam gawk
2. Execute the TCL Simulation Script: 
cd simulation/tcl
ns normal_traffic.tcl

This creates normal_traffic.tr (trace file) and normal_traffic.nam.

3. Process Trace File with AWK: 
awk -f ../awk/throughput.awk normal_traffic.tr
awk -f ../awk/pdr.awk normal_traffic.tr
awk -f ../awk/loss.awk normal_traffic.tr

AWK Processing Scripts

throughput.awk
BEGIN { received_bytes = 0; start_time = 1e9; end_time = 0; }
{
  event = $1; time = $2; pkt_size = $6;
  if (event == "r" || event == "-") {
    if (time < start_time) start_time = time;
    if (time > end_time) end_time = time;
  }
  if (event == "r") received_bytes += pkt_size;
}
END {
  duration = end_time - start_time;
  if (duration > 0) printf("Throughput: %.2f Kbps\n", (received_bytes * 8) / (duration * 1000));
}
loss.awk
BEGIN { sent = 0; dropped = 0; }
{
  event = $1;
  if (event == "+") sent++;
  if (event == "d") dropped++;
}
END {
  if (sent > 0) printf("Packet Loss Ratio: %.2f %%\n", (dropped / sent) * 100);
}
